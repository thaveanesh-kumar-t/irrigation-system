Required ESP32 Arduino Code Setup (JSON & CORS)

To ensure this web dashboard can fetch telemetry without being blocked by browser safety rules, add a /data endpoint with CORS headers in your ESP32 code:

// Add this route handler to your ESP32 web server code:
void handleData() {
  String json = "{";
  json += "\"temp\":" + String(temperature, 1) + ",";
  json += "\"hum\":" + String(humidity, 0) + ",";
  json += "\"moisture\":" + String(moisture) + ",";
  json += "\"pump\":" + String(pumpState ? "true" : "false") + ",";
  json += "\"mode\":\"" + String(operatingMode == 0 ? "AUTO" : (operatingMode == 1 ? "MANUAL ON" : "MANUAL OFF")) + "\"";
  json += "}";
  
  server.sendHeader("Access-Control-Allow-Origin", "*");
  server.send(200, "application/json", json);
}

// In setup():
server.on("/data", handleData);
server.enableCORS(true);
